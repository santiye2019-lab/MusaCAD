cmake_minimum_required(VERSION 3.22)
project(musacad_firstparty_dwg1032_cursor LANGUAGES C)
# Standalone: never link the legacy native core or copyleft dependencies.
add_library(musacad_firstparty_dwg1032_cursor STATIC src/musa_dwg1032_cursor.c src/musa_dwg1032_signature.c)
target_compile_features(musacad_firstparty_dwg1032_cursor PUBLIC c_std_11)
target_include_directories(musacad_firstparty_dwg1032_cursor PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
if(MSVC)
    target_compile_options(musacad_firstparty_dwg1032_cursor PRIVATE /W4 /WX)
else()
    target_compile_options(musacad_firstparty_dwg1032_cursor PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(musacad_firstparty_dwg1032_cursor_test tests/dwg1032_cursor_test.c)
    target_link_libraries(musacad_firstparty_dwg1032_cursor_test PRIVATE musacad_firstparty_dwg1032_cursor)
    add_test(NAME firstparty_dwg1032_bounded_binary_cursor COMMAND musacad_firstparty_dwg1032_cursor_test)
    add_executable(musacad_firstparty_dwg1032_u64_test tests/dwg1032_u64_test.c)
    target_link_libraries(musacad_firstparty_dwg1032_u64_test PRIVATE musacad_firstparty_dwg1032_cursor)
    add_test(NAME firstparty_dwg1032_64bit_fields COMMAND musacad_firstparty_dwg1032_u64_test)
    add_executable(musacad_firstparty_dwg1032_signature_test tests/dwg1032_signature_test.c)
    target_link_libraries(musacad_firstparty_dwg1032_signature_test PRIVATE musacad_firstparty_dwg1032_cursor)
    add_test(NAME firstparty_dwg1032_signature_preflight COMMAND musacad_firstparty_dwg1032_signature_test)
endif()
