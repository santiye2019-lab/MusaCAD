cmake_minimum_required(VERSION 3.22)
project(musacad_firstparty_dwg_route LANGUAGES C)
# Standalone, MIT and format-only: no GPL core, LibreDWG or production JNI.
add_library(musacad_firstparty_dwg_route STATIC src/musa_dwg_route.c)
target_include_directories(musacad_firstparty_dwg_route PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_compile_features(musacad_firstparty_dwg_route PUBLIC c_std_11)
if(MSVC)
    target_compile_options(musacad_firstparty_dwg_route PRIVATE /W4 /WX)
else()
    target_compile_options(musacad_firstparty_dwg_route PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(musacad_dwg_route_test tests/dwg_route_test.c)
    target_link_libraries(musacad_dwg_route_test PRIVATE musacad_firstparty_dwg_route)
    add_test(NAME firstparty_dwg_multi_version_route_candidate COMMAND musacad_dwg_route_test)
endif()
