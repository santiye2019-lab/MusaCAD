cmake_minimum_required(VERSION 3.22)
project(musacad_independent_scene2d LANGUAGES C)
# Intentionally standalone: NEVER link GPL/AGPL/LGPL legacy CAD sources.
add_library(musacad_independent_scene2d STATIC src/musa_scene2d.c)
target_compile_features(musacad_independent_scene2d PUBLIC c_std_11)
target_include_directories(musacad_independent_scene2d PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
if(MSVC)
    target_compile_options(musacad_independent_scene2d PRIVATE /W4 /WX)
else()
    target_compile_options(musacad_independent_scene2d PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(musacad_independent_scene2d_test tests/scene2d_test.c)
    target_link_libraries(musacad_independent_scene2d_test PRIVATE musacad_independent_scene2d)
    if(NOT MSVC)
        target_link_libraries(musacad_independent_scene2d_test PRIVATE m)
    endif()
    add_test(NAME independent_scene2d_graph_color_transform COMMAND musacad_independent_scene2d_test)
endif()
